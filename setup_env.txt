# 1. 가상환경 생성 및 활성화 (이미 하셨다면 건너뛰세요)
py -3.12 -m venv .venv
.\.venv\Scripts\activate

# 2. pip 최신화
python -m pip install --upgrade pip

# 3. [중요] CUDA 전용 PyTorch 우선 설치 
# 이 과정이 requirements.txt 설치보다 먼저 와야 합니다.
pip install torch>=2.4.0 torchvision>=0.19.0 torchaudio>=2.4.0 --index-url https://download.pytorch.org/whl/cu124

# 4. 나머지 라이브러리 설치
# --no-deps 옵션을 쓰지 않아도 이미 깔린 torch 버전을 인지하고 중복 설치를 피합니다.
pip install -r requirements.txt

# 5. 최종 GPU 작동 확인
python -c "import torch; print(f'GPU Check: {torch.cuda.is_available()} ({torch.cuda.get_device_name(0)})')"
